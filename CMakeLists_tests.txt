# NOTE on gating: these four binaries return non-zero when a case fails, so the
# exit code is the reliable signal. They used to carry a PASS_REGULAR_EXPRESSION

# "Results: 641 tests, 10 failures". A run with 10, 20 or 100 failures was

option(BVNR_FUZZ_TEST
    "Build the self-contained fuzz test and register it as CTest tests"
    ON)

set(BVNR_FUZZ_SMOKE_ITER 5000
    CACHE STRING "Mutation rounds for the fast CTest smoke fuzz tests")
set(BVNR_FUZZ_DEEP_ITER 100000
    CACHE STRING "Mutation rounds for the slow CTest deep fuzz tests")
set(BVNR_FUZZ_THOROUGH_ITER 200000
    CACHE STRING "Mutation rounds for the dedicated reader thorough fuzz test")

if (BVNR_FUZZ_TEST)

    add_executable(bvnr_fuzz_test tests/bovnar_fuzz_test.c)
    target_include_directories(bvnr_fuzz_test
        PRIVATE
            ${CMAKE_CURRENT_SOURCE_DIR}/include
            ${CMAKE_CURRENT_SOURCE_DIR}/src/utils)
    target_link_libraries(bvnr_fuzz_test PRIVATE bvnr_static)

    target_compile_options(bvnr_fuzz_test PRIVATE
        -Wall -Wextra -g
        $<$<AND:$<NOT:$<BOOL:${WIN32}>>,$<OR:$<CONFIG:Debug>,$<CONFIG:RelWithDebInfo>>>:
            -fsanitize=address,undefined
            -fno-omit-frame-pointer>)

    if (CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" AND NOT WIN32)
        target_link_options(bvnr_fuzz_test PRIVATE
            $<$<OR:$<CONFIG:Debug>,$<CONFIG:RelWithDebInfo>>:
                -fsanitize=address,undefined>)
    endif()

    macro(_bvnr_fuzz_test name harness iterations label timeout)
        add_test(
            NAME    ${name}
            COMMAND bvnr_fuzz_test
                        --harness    ${harness}
                        --iterations ${iterations}
                        --seed       42
        )
        set_tests_properties(${name} PROPERTIES
            TIMEOUT ${timeout}
            LABELS  "${label}"
            PASS_REGULAR_EXPRESSION "\\[bvnr_fuzz\\] PASS"

        )
    endmacro()

    _bvnr_fuzz_test(bvnr_fuzz_reader reader ${BVNR_FUZZ_SMOKE_ITER} "fuzz;fuzz_smoke" 60)
    _bvnr_fuzz_test(bvnr_fuzz_dom    dom    ${BVNR_FUZZ_SMOKE_ITER} "fuzz;fuzz_smoke" 60)
    _bvnr_fuzz_test(bvnr_fuzz_utils  utils  ${BVNR_FUZZ_SMOKE_ITER} "fuzz;fuzz_smoke" 60)

    _bvnr_fuzz_test(bvnr_fuzz_reader_deep    reader ${BVNR_FUZZ_DEEP_ITER}     "fuzz_deep"     600)
    _bvnr_fuzz_test(bvnr_fuzz_reader_thorough reader ${BVNR_FUZZ_THOROUGH_ITER} "fuzz_thorough" 300)
    _bvnr_fuzz_test(bvnr_fuzz_dom_deep        dom    ${BVNR_FUZZ_DEEP_ITER}     "fuzz_deep"     600)
    _bvnr_fuzz_test(bvnr_fuzz_utils_deep      utils  ${BVNR_FUZZ_DEEP_ITER}     "fuzz_deep"     600)

    add_executable(bvnr_fuzz_writer_test tests/bovnar_fuzz_writer.c)
    target_compile_definitions(bvnr_fuzz_writer_test PRIVATE FUZZ_STANDALONE)
    target_include_directories(bvnr_fuzz_writer_test
        PRIVATE
            ${CMAKE_CURRENT_SOURCE_DIR}/include
            ${CMAKE_CURRENT_SOURCE_DIR}/src/utils)
    target_link_libraries(bvnr_fuzz_writer_test PRIVATE bvnr_static)

    target_compile_options(bvnr_fuzz_writer_test PRIVATE
        -Wall -Wextra -g
        $<$<AND:$<NOT:$<BOOL:${WIN32}>>,$<OR:$<CONFIG:Debug>,$<CONFIG:RelWithDebInfo>>>:
            -fsanitize=address,undefined
            -fno-omit-frame-pointer>)

    if (CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" AND NOT WIN32)
        target_link_options(bvnr_fuzz_writer_test PRIVATE
            $<$<OR:$<CONFIG:Debug>,$<CONFIG:RelWithDebInfo>>:
                -fsanitize=address,undefined>)
    endif()

    # TWO seeds, deliberately. The original seed was the single byte "X", which
    # sounds useless -- a mutator given one byte only reaches one-byte
    # neighbourhoods, so the harness never saw an annotation, a unit, an array or
    # a struct. But measuring beats assuming: an injected 8-byte overrun in
    # push_mem is caught from the "X" seed at 1000 iterations and NOT caught from
    # the rich document even at 20 000, because the two drive completely
    # different output-length distributions and the bug needs a near-full sink
    # buffer. Keep both rather than trading one blind spot for another.
    set(_bvnr_writer_seed_min "${CMAKE_BINARY_DIR}/bvnr_fuzz_writer_seed_min.bin")
    file(WRITE "${_bvnr_writer_seed_min}" "X")

    set(_bvnr_writer_seed "${CMAKE_BINARY_DIR}/bvnr_fuzz_writer_seed.bvnr")
    file(WRITE "${_bvnr_writer_seed}"
"#!bovnar 1.1\n"
".name = \"seed\";\n"
".port = <uint:16> 443;\n"
".ratio = <sint:32> -7;\n"
".speed = <float:64,k~m/h> 88.5;\n"
".hex = <uint:32,_16> \"deadbeef\";\n"
".price = <float_dec:64,$USD> 19.99;\n"
".when = <datetime:64,tai> 2016-12-31T23:59:60.5Z;\n"
".flag = on;\n"
".sym = ok;\n"
".ref = &.name;\n"
".grid = [1, 2, 3]/[4, 5, 6];\n"
".rec = {.a = 1; .b = \"two\";};\n"
".hole = ;\n"
".esc = \"tab\\there caf\\u{e9}\";\n")

    # Iteration counts. These used to be 25/30/25 with timeouts of 60/600/300 s,
    # i.e. 0.01 s of actual work each -- and the comment justifying the cap
    # claimed --iterations 1000 "reliably trips the __builtin_trap() invariants".
    # That is stale: 200 000 iterations pass cleanly on seeds 42, 7 and 999, and
    # take 1.7 s. The cap bought nothing and cost the writer its only oracle --
    # an injected heap overflow in push_mem survived all three tiers and was
    # caught at 1000. Each tier also uses its own seed now; all three shared 42,
    # so the deep and thorough runs were re-testing the smoke run's inputs.
    add_test(NAME bvnr_fuzz_writer
        COMMAND bvnr_fuzz_writer_test --iterations 20000 --seed 42
                "${_bvnr_writer_seed_min}")
    set_tests_properties(bvnr_fuzz_writer PROPERTIES
        TIMEOUT 60
        LABELS  "fuzz;fuzz_smoke"
        PASS_REGULAR_EXPRESSION "\\[bvnr_fuzz_writer\\] PASS")

    add_test(NAME bvnr_fuzz_writer_deep
        COMMAND bvnr_fuzz_writer_test --iterations 200000 --seed 999
                "${_bvnr_writer_seed_min}")
    set_tests_properties(bvnr_fuzz_writer_deep PROPERTIES
        TIMEOUT 600
        LABELS  "fuzz_deep"
        PASS_REGULAR_EXPRESSION "\\[bvnr_fuzz_writer\\] PASS")

    add_test(NAME bvnr_fuzz_writer_thorough
        COMMAND bvnr_fuzz_writer_test --iterations 100000 --seed 7
                "${_bvnr_writer_seed_min}")
    set_tests_properties(bvnr_fuzz_writer_thorough PROPERTIES
        TIMEOUT 300
        LABELS  "fuzz_thorough"
        PASS_REGULAR_EXPRESSION "\\[bvnr_fuzz_writer\\] PASS")

    # The document seed, over its own distribution.
    add_test(NAME bvnr_fuzz_writer_doc
        COMMAND bvnr_fuzz_writer_test --iterations 100000 --seed 4242

    set_tests_properties(bvnr_fuzz_writer_doc PROPERTIES

        PASS_REGULAR_EXPRESSION "\\[bvnr_fuzz_writer\\] PASS")

endif()

    set(BVNR_FUZZ_ENGINE "libfuzzer" CACHE STRING
        "Fuzzing engine for external targets: libfuzzer | afl")
    set_property(CACHE BVNR_FUZZ_ENGINE PROPERTY STRINGS libfuzzer afl)

    set(BVNR_FUZZ_SANITIZERS "address;undefined" CACHE STRING

    # Sanitizer / coverage flags.  The library translation units are
    # instrumented for coverage feedback but must NOT pull in libFuzzer's
    # main() — only the harness target links the fuzzer runtime, hence
    # "fuzzer-no-link" on the library and "fuzzer" on the harness.  Under
    # AFL the compiler driver (afl-clang-fast) performs instrumentation, so
    # the library and harness share one flag set and the source's own
    # __AFL_LOOP main() is compiled in.
    if (BVNR_FUZZ_ENGINE STREQUAL "libfuzzer")
        set(_ext_lib_cf -fsanitize=fuzzer-no-link,${_ext_san_str})
        set(_ext_cf     -fsanitize=fuzzer,${_ext_san_str})
        set(_ext_lf     -fsanitize=fuzzer,${_ext_san_str})
    else()
        set(_ext_lib_cf -fsanitize=${_ext_san_str})
        set(_ext_cf     -fsanitize=${_ext_san_str})
        set(_ext_lf     -fsanitize=${_ext_san_str})
    endif()

    list(APPEND _ext_lib_cf -g -O1 -fno-omit-frame-pointer)
    list(APPEND _ext_cf     -g -O1 -fno-omit-frame-pointer)

    # Instrumented copy of the core library.  Kept separate from
    # bvnr_static (built without instrumentation for the normal unit tests)
    # so the fuzzers get coverage feedback through the parser internals.
    add_library(bvnr_fuzz_objects OBJECT ${BVNR_SOURCES})
    target_compile_options(bvnr_fuzz_objects PRIVATE
        -Wall -Wextra ${_ext_lib_cf})
    target_include_directories(bvnr_fuzz_objects
        PUBLIC  ${CMAKE_CURRENT_SOURCE_DIR}/include
        PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src/io
                ${CMAKE_CURRENT_SOURCE_DIR}/src/lexer
                ${CMAKE_CURRENT_SOURCE_DIR}/src/utils)

    # _bvnr_fuzz_external(<name> <source>): a coverage-guided fuzz target.
    # libFuzzer supplies main(); under AFL the source's __AFL_LOOP main() is
    # compiled in by the afl driver.  These are run manually, not by CTest:
    #   ./build/tests/bvnr_fuzz_ext_reader -max_total_time=60 corpus/reader
    macro(_bvnr_fuzz_external name source)
        add_executable(bvnr_fuzz_ext_${name} ${source}
            $<TARGET_OBJECTS:bvnr_fuzz_objects>)
        target_compile_options(bvnr_fuzz_ext_${name} PRIVATE
            -Wall -Wextra ${_ext_cf})
        target_include_directories(bvnr_fuzz_ext_${name} PRIVATE
            ${CMAKE_CURRENT_SOURCE_DIR}/include
            ${CMAKE_CURRENT_SOURCE_DIR}/src/utils)
        target_link_options(bvnr_fuzz_ext_${name} PRIVATE ${_ext_lf})
    endmacro()

    _bvnr_fuzz_external(reader tests/bovnar_fuzz_reader.c)
    _bvnr_fuzz_external(dom    tests/bovnar_fuzz_dom.c)
    _bvnr_fuzz_external(utils  tests/bovnar_fuzz_utils.c)

    # Corpus seed generator: writes starter inputs under
    # <out>/{reader,dom,utils} (default ./corpus).  Plain tool, no
    # instrumentation.  Run before the fuzzers to seed their corpora.
    add_executable(bvnr_fuzz_seed_gen tests/bovnar_fuzz_seed_gen.c)
    target_compile_options(bvnr_fuzz_seed_gen PRIVATE
        -Wall -Wextra)

endif()

# ── CLI example-file smoke tests ─────────────────────────────────────────────
#
# For every .bvnr file under examples/ run two commands against the built
# bovnar binary:
#
#   bovnar events  <file>   – must exit 0 and print the validated-ok line
#   bovnar validate <file>  – must exit 0
#
# The macro _bvnr_cli_example(stem) derives the file path from the stem name
# and registers both tests.  Each test depends on the bovnar target so that
# CTest knows to build it first when run in isolation.

macro(_bvnr_cli_example stem)
    set(_bvnr_ex_file "${CMAKE_CURRENT_SOURCE_DIR}/examples/${stem}.bvnr")

    add_test(
        NAME    bvnr_cli_events_${stem}
        COMMAND bovnar events "${_bvnr_ex_file}"
    )
    set_tests_properties(bvnr_cli_events_${stem} PROPERTIES
        LABELS                "cli;examples;events"
        TIMEOUT               30
        PASS_REGULAR_EXPRESSION "All tokens validated successfully"
    )

    add_test(
        NAME    bvnr_cli_validate_${stem}
        COMMAND bovnar validate "${_bvnr_ex_file}"
    )
    set_tests_properties(bvnr_cli_validate_${stem} PROPERTIES
        LABELS  "cli;examples;validate"
        TIMEOUT 30
        PASS_REGULAR_EXPRESSION ": OK"
        FAIL_REGULAR_EXPRESSION "Validation failed|error|Error")

    # Assert the canonical serialiser is idempotent: pretty-printing the
    # canonical output again must reproduce it byte-for-byte (and re-parse).
    add_test(
        NAME    bvnr_cli_idempotent_${stem}
        COMMAND "${CMAKE_COMMAND}"
                -DBOVNAR=$<TARGET_FILE:bovnar> -DEMULATOR=${CMAKE_CROSSCOMPILING_EMULATOR}

                -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/pretty_print_idempotent.cmake)
    set_tests_properties(bvnr_cli_idempotent_${stem} PROPERTIES
        LABELS  "cli;examples;idempotent"
        TIMEOUT 30)
endmacro()

        COMMAND ${Python3_EXECUTABLE} -c "import markdown"

            "markdown not importable by ${Python3_EXECUTABLE} -- skipping the "
            "generated HTML doc check. Run 'pip install markdown' to enable it.")
    endif()

# ── canonical document form ──────────────────────────────────────────────────
# `pretty-print --canonical` claims something pretty-printing does not: that two
# documents which MEAN the same thing produce the same bytes. That is what a
# signature, a content hash or a semantic diff needs, and it is testable in
# exactly one way -- feed it two spellings of one document and compare.
#
# Idempotence is checked in the same run, because a "canonical" form that moves
# on a second pass is just one more spelling and would break any consumer storing
# a hash of it.
add_test(
    NAME    bvnr_cli_canonical_converges
    COMMAND "${CMAKE_COMMAND}"
            -DBOVNAR=$<TARGET_FILE:bovnar> -DEMULATOR=${CMAKE_CROSSCOMPILING_EMULATOR}
            -DFILE_A=${CMAKE_CURRENT_SOURCE_DIR}/tests/canonical_prefixed.bvnr
            -DFILE_B=${CMAKE_CURRENT_SOURCE_DIR}/tests/canonical_reduced.bvnr
            -DTMP_DIR=${CMAKE_CURRENT_BINARY_DIR}/canonical
            -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/canonical_converges.cmake)
set_tests_properties(bvnr_cli_canonical_converges PROPERTIES
    LABELS "cli;canonical;units" TIMEOUT 30)

# ── the IETF draft against the header ────────────────────────────────────────
# doc/ietf/draft-sonntag-bovnar-00 is a SECOND, independent statement of the
# format, rendered into four committed artifacts, and nothing compared it to
# anything: bvnr_doc_refs checks that citations resolve and that the spec's error
# table matches the header, bvnr_doc_layout checks structure, and neither looks
# at the draft or at any numeric CLAIM.
#
# So it drifted, and doc/05 with it: both said a compound unit may not exceed
# "8 components" long after BVNR_MAX_UNIT_COMPONENTS became 32, while doc/03 said
# 32. Someone implementing from the draft would have rejected documents the
# reference implementation accepts -- an interoperability defect in the one
# document whose whole purpose is interoperability.
#
# The renderings are checked too, not just the .md they are generated from:
# kramdown-rfc and xml2rfc are not run here or in CI, so a fix applied to the
# source alone would leave three committed files stating the old value.
if(Python3_Interpreter_FOUND AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/check_ietf_draft.py)
    add_test(NAME bvnr_ietf_draft
             COMMAND ${Python3_EXECUTABLE}
                     ${CMAKE_CURRENT_SOURCE_DIR}/check_ietf_draft.py)
    set_tests_properties(bvnr_ietf_draft PROPERTIES LABELS "docs;ietf" TIMEOUT 60)
endif()

# ── the profile tables against the publications they transliterate ───────────
# The one thing nothing else covers: every other check proves the tables are
# self-consistent, and five tables wrong in the same way would agree with each
# other perfectly (doc/11 §14.3). This reads UCUM's and UDUNITS' own machine-
# readable definitions and compares each mapped row's factor and dimension
# against what the reference library says the native target is worth.
#
# It SKIPS, green, when the publisher files are not cached: a test must not
# depend on the network, which is the same rule bvnr_web_links follows. Populate
# the cache once with
#     python3 check_profile_factors.py --fetch
# and every later run checks for real. CI should fetch and then use --strict so
# that a skip there is a failure rather than a quiet pass.
#
# Labelled "python" although the name does not start with bvnr_py_: the label
# records "drives libbvnr from a Python interpreter", and LIBBOVNAR_DIR below is
# exactly that -- the checker resolves every native target through the ctypes
# bindings. A sanitizer build cannot do it (ASan refuses a dlopen of its own
# runtime into an uninstrumented interpreter: "ASan runtime does not come first
# in initial library list"), so the ASan pass excludes it with -LE python, the
# same way bvnr_data_parity is excluded. The unsanitized pass still runs it.
if(Python3_Interpreter_FOUND
   AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/check_profile_factors.py)
    add_test(NAME bvnr_profile_factors
             COMMAND ${Python3_EXECUTABLE}
                     ${CMAKE_CURRENT_SOURCE_DIR}/check_profile_factors.py
                     --cache ${CMAKE_CURRENT_BINARY_DIR}/vocab)
    set_tests_properties(bvnr_profile_factors PROPERTIES
        LABELS "python;units;profile;vocab" TIMEOUT 120
        ENVIRONMENT "LIBBOVNAR_DIR=${CMAKE_CURRENT_BINARY_DIR}")
endif()

# ── Website translations still match the English source ─────────────────────
# gen_i18n.py keys each translation on the English text it was made from, so
# editing web/index.html orphans the German for that string. Nothing about the
# English page changes when that happens -- it keeps working perfectly -- which
# is exactly why this needs a test: the failure is invisible until a German
# visitor reads a paragraph that silently reverted to English. --check also
# rejects a document-relative asset path added to a <script>, which would 404
# only on the translated pages.
if(Python3_Interpreter_FOUND AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/web/i18n)
    # CONFIGURE_DEPENDS so dropping in web/i18n/fr.json re-runs CMake and
    # actually registers its gate, instead of waiting for an unrelated
    # reconfigure. Note a new language file also moves the registered test
    # count, so BVNR_EXPECTED_TEST_COUNT must be bumped with it.
    file(GLOB _i18n_tables CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/web/i18n/*.json)
    foreach(_tbl ${_i18n_tables})
        get_filename_component(_lang "${_tbl}" NAME_WE)
        add_test(
            NAME    bvnr_i18n_${_lang}
            COMMAND ${Python3_EXECUTABLE}
                    ${CMAKE_CURRENT_SOURCE_DIR}/gen_i18n.py --check ${_lang})
        set_tests_properties(bvnr_i18n_${_lang} PROPERTIES
            LABELS "web;i18n" TIMEOUT 60)
    endforeach()
endif()

# ── Amalgamation reproducibility ────────────────────────────────────────────
# bvnr_wasm_freshness above hashes the amalgamated sources, so it is only a
# freshness check if that hash depends on the SOURCES and nothing else. Twice now
# it has also depended on the build directory's name, and the second time the
# suite was green in build/ while failing in build-asan/ over identical inputs --
# a gate accusing the tree of being stale when the tree was fine. Pin the
# property the gate rests on rather than rediscovering it from the wreckage.
if(Python3_Interpreter_FOUND AND NOT WIN32)
    add_test(
        NAME    bvnr_amalgam_reproducible
        COMMAND "${CMAKE_COMMAND}"
                -DPYTHON=${Python3_EXECUTABLE}
                -DSCRIPT=${CMAKE_CURRENT_SOURCE_DIR}/amalgamate.py
                -DGENERATED=${CMAKE_CURRENT_BINARY_DIR}/generated
                -DWORKDIR=${CMAKE_CURRENT_BINARY_DIR}/amalgam_repro
                -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/amalgam_reproducible.cmake)
    set_tests_properties(bvnr_amalgam_reproducible PROPERTIES
        LABELS "wasm;packaging;codegen" TIMEOUT 120)
endif()

# ── Single-file amalgamation ────────────────────────────────────────────────
# dist/bovnar.{c,h} are git-ignored, so the FRESHNESS half of this test (does the
# committed amalgamation match the sources?) has nothing to compare against and
# the driver skips it. The other half is not conditional on shipping anything and
# was disabled along with it, which left the amalgamation with no compile
# coverage at all: bvnr_amalgam_reproducible above proves only that two
# generations are byte-identical, which a file that does not compile satisfies
# perfectly.
#
# That gap has a cost the rest of the suite cannot see. An integrator's first act
# with a single-file library is to drop bovnar.c into their build, under their
# own warning flags. Ours were hiding a real one: -Wno-override-init in
# BVNR_C_FLAGS silenced 546 deliberate designated-initialiser overrides in the
# lexer state table for THIS build only, and the amalgamation carries that table
# verbatim -- so their first build produced 546 warnings ours never showed. The
# suppression now lives in a pragma in src/lexer/bovnar_state_table.c, where it
# travels with the code, and this test is what keeps it true: strict flags,
# -Werror, and NO -Wno- of any kind, which is the integrator's build and not
# ours.
if(Python3_Interpreter_FOUND AND NOT WIN32 AND NOT BVNR_SANITIZE
   AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tests/amalgam_smoke.c)
    add_test(
        NAME    bvnr_amalgam_smoke
        COMMAND "${CMAKE_COMMAND}"
                -DSRC_DIR=${CMAKE_CURRENT_SOURCE_DIR}
                -DPYTHON=${Python3_EXECUTABLE}
                -DCC=${CMAKE_C_COMPILER}
                -DBIN_DIR=${CMAKE_CURRENT_BINARY_DIR}/amalgam_smoke
                -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/amalgam_smoke.cmake)
    set_tests_properties(bvnr_amalgam_smoke PROPERTIES
        LABELS "packaging;codegen;amalgam" TIMEOUT 300)
endif()

# ── Registration self-check ─────────────────────────────────────────────────
# Only meaningful for a default configure; any option that adds or removes tests
# makes the count legitimately differ, so this compares only when the relevant
# knobs are at their defaults. _bvnr_have_pytest, not Python3_Interpreter_FOUND:
# the Python gates now need the module too, so an interpreter without it is a
# legitimate shortfall of 20 (the 19 bvnr_py_* gates plus the bvnr_abi_dump_build
# fixture registered beside them) rather than something silently unregistering.
get_property(_bvnr_all_tests DIRECTORY PROPERTY TESTS)
list(LENGTH _bvnr_all_tests _bvnr_ntests)
if(BVNR_PACKAGE AND BVNR_FUZZ_TEST AND NOT BVNR_SANITIZE AND NOT WIN32
   AND NOT CMAKE_CROSSCOMPILING AND _bvnr_have_pytest)
    if(NOT _bvnr_ntests EQUAL ${BVNR_EXPECTED_TEST_COUNT})
        message(FATAL_ERROR
            "registered ${_bvnr_ntests} tests, expected "
            "${BVNR_EXPECTED_TEST_COUNT}. If you added or removed one, update "
            "BVNR_EXPECTED_TEST_COUNT in CMakeLists_tests.txt. If you did not, a "
            "test has silently stopped registering -- check the conditions "
            "guarding it.")
    endif()
endif()
message(STATUS "registered ${_bvnr_ntests} tests")
